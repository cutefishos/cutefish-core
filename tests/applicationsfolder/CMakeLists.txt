cmake_minimum_required(VERSION 3.22)
project(applications_folder_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CutefishToolkit REQUIRED)
find_package(GTest REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(Systemd REQUIRED IMPORTED_TARGET libsystemd)
set(SERVICE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../applicationsfolder)
add_executable(applications_folder_test applications_folder_test.cpp ${SERVICE_DIR}/applicationsfolderservice.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/../../common/detached_call.cpp)
target_include_directories(applications_folder_test PRIVATE ${SERVICE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../..)
target_link_libraries(applications_folder_test PRIVATE Cutefish::ToolkitBase PkgConfig::Systemd GTest::gtest_main)
enable_testing()
add_test(NAME applications_folder_test COMMAND applications_folder_test)
